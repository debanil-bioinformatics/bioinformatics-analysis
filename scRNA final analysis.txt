# ==============================================================================
# CUEDC2 vs ESTROGEN RESPONSE VALIDATION - EPITHELIAL scRNA-seq (TNBC)
# ==============================================================================
# Mirrors the bulk METABRIC/TCGA methodology:
#   - ssGSEA scores (Hallmark Estrogen Response Early/Late) per cell
#   - Spearman correlation: CUEDC2 expression vs ssGSEA score
#   - Median-split (per sample) + Wilcoxon test, BH-FDR corrected
# Input:  Wu_et_al_TNBC_epithelial.rds (15,428 epithelial cells, 8 TNBC samples)
# ==============================================================================

library(tidyverse)
library(Seurat)
library(GSVA)
library(msigdbr)

set.seed(42)

wu_dir     <- path.expand("~/Documents/Cancers/metadata/GSE161529_Wu_et_al_TNBC")
input_rds  <- file.path(wu_dir, "processed", "Wu_et_al_TNBC_epithelial.rds")
output_dir <- file.path(wu_dir, "processed")
figure_dir <- file.path(wu_dir, "figures")

cat("\n", strrep("=", 100), "\n")
cat("CUEDC2 vs ESTROGEN RESPONSE - EPITHELIAL SINGLE-CELL VALIDATION\n")
cat(strrep("=", 100), "\n\n")

# ==============================================================================
# STEP 1: LOAD EPITHELIAL OBJECT
# ==============================================================================

cat("Loading epithelial subset...\n")
scRNA <- readRDS(input_rds)
cat("Cells:", ncol(scRNA), " Genes:", nrow(scRNA), "\n\n")

gene_map     <- scRNA@misc$gene_mapping
symbol_to_id <- setNames(names(gene_map), gene_map)

get_id <- function(symbol) {
  id <- symbol_to_id[symbol]
  id <- id[!is.na(id) & id %in% rownames(scRNA)]
  if (length(id) == 0) return(NA)
  id[1]
}

# ==============================================================================
# STEP 2: LOCATE CUEDC2 AND CONFIRM IN DATASET
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 2: LOCATING CUEDC2\n")
cat(strrep("-", 100), "\n\n")

cuedc2_id <- get_id("CUEDC2")
if (is.na(cuedc2_id)) stop("CUEDC2 not found in dataset!")
cat("CUEDC2 ENSG ID:", cuedc2_id, "\n")

# Pull normalized (log-normalized) expression - already computed in earlier step
cuedc2_expr <- FetchData(scRNA, vars = cuedc2_id, layer = "data")[, 1]
cat("CUEDC2 expression summary (log-normalized):\n")
print(summary(cuedc2_expr))
cat("Cells with detectable CUEDC2 (>0):",
    sum(cuedc2_expr > 0), "/", length(cuedc2_expr),
    sprintf("(%.1f%%)", 100 * mean(cuedc2_expr > 0)), "\n\n")

scRNA$CUEDC2_expr <- cuedc2_expr

# ==============================================================================
# STEP 3: CONFIRM ESR1 DETECTABILITY (for context, not used in ssGSEA directly)
# ==============================================================================

esr1_id <- get_id("ESR1")
if (!is.na(esr1_id)) {
  esr1_expr <- FetchData(scRNA, vars = esr1_id, layer = "data")[, 1]
  cat("ESR1 detectable in",
      sum(esr1_expr > 0), "/", length(esr1_expr),
      sprintf("(%.1f%%) epithelial cells\n\n", 100 * mean(esr1_expr > 0)))
  scRNA$ESR1_expr <- esr1_expr
} else {
  cat("ESR1 not found under that symbol in features file.\n\n")
}

# ==============================================================================
# STEP 4: GET HALLMARK ESTROGEN RESPONSE GENE SETS (msigdbr) AND MAP TO ENSG
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 4: LOADING HALLMARK ESTROGEN RESPONSE GENE SETS\n")
cat(strrep("-", 100), "\n\n")

hallmark <- msigdbr(species = "Homo sapiens", collection = "H")

early_symbols <- hallmark %>%
  filter(gs_name == "HALLMARK_ESTROGEN_RESPONSE_EARLY") %>%
  pull(gene_symbol) %>% unique()

late_symbols <- hallmark %>%
  filter(gs_name == "HALLMARK_ESTROGEN_RESPONSE_LATE") %>%
  pull(gene_symbol) %>% unique()

early_ids <- na.omit(sapply(early_symbols, get_id))
late_ids  <- na.omit(sapply(late_symbols, get_id))

cat("Estrogen Response Early: ", length(early_ids), "/", length(early_symbols),
    " genes mapped and present in dataset\n", sep = "")
cat("Estrogen Response Late:  ", length(late_ids), "/", length(late_symbols),
    " genes mapped and present in dataset\n\n", sep = "")

gene_sets <- list(
  Estrogen_Response_Early = as.character(early_ids),
  Estrogen_Response_Late  = as.character(late_ids)
)

# ==============================================================================
# STEP 5: ssGSEA SCORING (per cell)
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 5: COMPUTING ssGSEA SCORES (per cell) - this may take several minutes\n")
cat(strrep("-", 100), "\n\n")

expr_mat <- GetAssayData(scRNA, assay = "RNA", layer = "data")

gsva_par <- ssgseaParam(
  expr = as.matrix(expr_mat),
  geneSets = gene_sets,
  normalize = TRUE
)

ssgsea_scores <- gsva(gsva_par, verbose = TRUE)
ssgsea_scores <- t(ssgsea_scores)  # cells x pathways

cat("ssGSEA scoring complete.\n\n")

scRNA$ER_Early_ssGSEA <- ssgsea_scores[, "Estrogen_Response_Early"]
scRNA$ER_Late_ssGSEA  <- ssgsea_scores[, "Estrogen_Response_Late"]

# ==============================================================================
# STEP 6: SPEARMAN CORRELATION - CUEDC2 vs ssGSEA SCORES (per sample + pooled)
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 6: SPEARMAN CORRELATION (CUEDC2 vs ER response)\n")
cat(strrep("-", 100), "\n\n")

meta <- scRNA@meta.data %>%
  select(sample, CUEDC2_expr, ER_Early_ssGSEA, ER_Late_ssGSEA) %>%
  mutate(sample = as.character(sample))

run_corr <- function(df, group_label) {
  early <- cor.test(df$CUEDC2_expr, df$ER_Early_ssGSEA, method = "spearman", exact = FALSE)
  late  <- cor.test(df$CUEDC2_expr, df$ER_Late_ssGSEA,  method = "spearman", exact = FALSE)
  tibble(
    Group = group_label,
    N = nrow(df),
    Early_rho = unname(early$estimate), Early_P = early$p.value,
    Late_rho  = unname(late$estimate),  Late_P  = late$p.value
  )
}

corr_pooled <- run_corr(meta, "Pooled_epithelial")
corr_per_sample <- meta %>%
  group_split(sample) %>%
  map_dfr(~ run_corr(.x, unique(.x$sample)))

corr_results <- bind_rows(corr_pooled, corr_per_sample) %>%
  mutate(
    Early_FDR = p.adjust(Early_P, method = "BH"),
    Late_FDR  = p.adjust(Late_P,  method = "BH")
  )

cat("Correlation results:\n")
print(as.data.frame(corr_results))
cat("\n")

write.csv(corr_results, file.path(output_dir, "scRNA_CUEDC2_ER_correlation.csv"), row.names = FALSE)
cat("Saved: scRNA_CUEDC2_ER_correlation.csv\n\n")

# ==============================================================================
# STEP 7: MEDIAN-SPLIT STRATIFICATION + WILCOXON TEST (per sample, pooled)
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 7: MEDIAN-SPLIT (CUEDC2 High/Low) + WILCOXON TEST\n")
cat(strrep("-", 100), "\n\n")

run_wilcox <- function(df, group_label) {
  med <- median(df$CUEDC2_expr)
  df$CUEDC2_group <- ifelse(df$CUEDC2_expr > med, "High", "Low")
  df <- df %>% filter(CUEDC2_group %in% c("High", "Low"))
  
  early_w <- wilcox.test(ER_Early_ssGSEA ~ CUEDC2_group, data = df)
  late_w  <- wilcox.test(ER_Late_ssGSEA  ~ CUEDC2_group, data = df)
  
  tibble(
    Group = group_label,
    N = nrow(df),
    N_High = sum(df$CUEDC2_group == "High"),
    N_Low  = sum(df$CUEDC2_group == "Low"),
    Median_Early_High = median(df$ER_Early_ssGSEA[df$CUEDC2_group == "High"]),
    Median_Early_Low  = median(df$ER_Early_ssGSEA[df$CUEDC2_group == "Low"]),
    Early_P = early_w$p.value,
    Median_Late_High = median(df$ER_Late_ssGSEA[df$CUEDC2_group == "High"]),
    Median_Late_Low  = median(df$ER_Late_ssGSEA[df$CUEDC2_group == "Low"]),
    Late_P = late_w$p.value
  )
}

wilcox_pooled <- run_wilcox(meta, "Pooled_epithelial")
wilcox_per_sample <- meta %>%
  group_split(sample) %>%
  map_dfr(~ run_wilcox(.x, unique(.x$sample)))

wilcox_results <- bind_rows(wilcox_pooled, wilcox_per_sample) %>%
  mutate(
    Early_FDR = p.adjust(Early_P, method = "BH"),
    Late_FDR  = p.adjust(Late_P,  method = "BH")
  )

cat("Median-split Wilcoxon results:\n")
print(as.data.frame(wilcox_results))
cat("\n")

write.csv(wilcox_results, file.path(output_dir, "scRNA_CUEDC2_ER_medianSplit_Wilcoxon.csv"), row.names = FALSE)
cat("Saved: scRNA_CUEDC2_ER_medianSplit_Wilcoxon.csv\n\n")

# ==============================================================================
# STEP 8: PUBLICATION-QUALITY FIGURES
# ==============================================================================

cat(strrep("-", 100), "\n")
cat("STEP 8: BUILDING FIGURES\n")
cat(strrep("-", 100), "\n\n")

pooled_row <- corr_results %>% filter(Group == "Pooled_epithelial")
