# --- 4e: Per-sample UCell correlation ---
run_ucell_corr <- function(df, group_label) {
  early <- cor.test(df$CUEDC2_expr, df$ER_Early_UCell, method = "spearman", exact = FALSE)
  late  <- cor.test(df$CUEDC2_expr, df$ER_Late_UCell,  method = "spearman", exact = FALSE)
  tibble(
    Method = paste0("UCell_", group_label),
    N = nrow(df),
    Early_rho = unname(early$estimate), Early_P = early$p.value,
    Late_rho  = unname(late$estimate),  Late_P  = late$p.value
  )
}

ucell_per_sample <- ucell_meta %>%
  group_split(sample) %>%
  map_dfr(~ run_ucell_corr(.x, unique(.x$sample)))

ucell_results <- bind_rows(ucell_pooled_corr, ucell_per_sample) %>%
  mutate(
    Early_FDR = p.adjust(Early_P, method = "BH"),
    Late_FDR  = p.adjust(Late_P,  method = "BH")
  )

# --- 4f: UCell median-split + Wilcoxon ---
run_ucell_wilcox <- function(df, group_label) {
  med <- median(df$CUEDC2_expr)
  df$CUEDC2_group <- ifelse(df$CUEDC2_expr > med, "High", "Low")
  df <- df %>% filter(CUEDC2_group %in% c("High", "Low"))

  early_w <- wilcox.test(ER_Early_UCell ~ CUEDC2_group, data = df)
  late_w  <- wilcox.test(ER_Late_UCell  ~ CUEDC2_group, data = df)

  tibble(
    Method = paste0("UCell_", group_label),
    N = nrow(df),
    N_High = sum(df$CUEDC2_group == "High"),
    N_Low  = sum(df$CUEDC2_group == "Low"),
    Median_Early_High = median(df$ER_Early_UCell[df$CUEDC2_group == "High"]),
    Median_Early_Low  = median(df$ER_Early_UCell[df$CUEDC2_group == "Low"]),
    Early_P = early_w$p.value,
    Median_Late_High = median(df$ER_Late_UCell[df$CUEDC2_group == "High"]),
    Median_Late_Low  = median(df$ER_Late_UCell[df$CUEDC2_group == "Low"]),
    Late_P = late_w$p.value
  )
}
