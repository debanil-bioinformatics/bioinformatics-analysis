# ==============================================================================
# QC DIAGNOSTIC: PER-CLUSTER MARKER SCORE HEATMAP
# ==============================================================================
# Shows all 7 cell-type module scores across all 24 clusters, so you can see
# the margin between the winning cell type and the runner-up for each cluster.
# ==============================================================================

library(tidyverse)
library(Seurat)
library(pheatmap)

wu_dir     <- path.expand("~/Documents/Cancers/metadata/GSE161529_Wu_et_al_TNBC")
input_rds  <- file.path(wu_dir, "processed", "Wu_et_al_TNBC_annotated.rds")
figure_dir <- file.path(wu_dir, "figures")
output_dir <- file.path(wu_dir, "processed")

cat("Loading annotated object...\n")
scRNA <- readRDS(input_rds)

# ==============================================================================
# COMPUTE CLUSTER-MEAN SCORES (same score columns as before)
# ==============================================================================

score_cols <- grep("^score_", colnames(scRNA@meta.data), value = TRUE)
cat("Score columns found:", paste(score_cols, collapse = ", "), "\n\n")

cluster_scores <- scRNA@meta.data %>%
  select(seurat_clusters, cell_type, all_of(score_cols)) %>%
  group_by(seurat_clusters, cell_type) %>%
  summarise(across(all_of(score_cols), mean), n_cells = n(), .groups = "drop") %>%
  arrange(as.numeric(as.character(seurat_clusters)))

cat("Cluster-level mean scores:\n")
print(as.data.frame(cluster_scores))
cat("\n")

# ==============================================================================
# BUILD MATRIX FOR HEATMAP (row = cluster, col = cell type score)
# ==============================================================================

score_mat <- as.matrix(cluster_scores[, score_cols])
rownames(score_mat) <- paste0(
  "C", cluster_scores$seurat_clusters,
  " (", cluster_scores$cell_type, ", n=", cluster_scores$n_cells, ")"
)
colnames(score_mat) <- sub("^score_", "", colnames(score_mat))

# Z-score across clusters (per cell-type signature) so the heatmap highlights
# RELATIVE enrichment per column, easier to visually confirm cluster identity
score_mat_z <- scale(score_mat)

# ==============================================================================
# MARGIN CHECK: winning score vs runner-up, per cluster
# ==============================================================================

margin_df <- as.data.frame(score_mat) %>%
  rownames_to_column("cluster_label") %>%
  pivot_longer(-cluster_label, names_to = "cell_type_score", values_to = "score") %>%
  group_by(cluster_label) %>%
  arrange(desc(score), .by_group = TRUE) %>%
  summarise(
    winner       = first(cell_type_score),
    winner_score = first(score),
    runnerup     = nth(cell_type_score, 2),
    runnerup_score = nth(score, 2),
    margin       = winner_score - runnerup_score,
    .groups = "drop"
  ) %>%
  arrange(margin)

cat("Clusters with SMALLEST margin (winner vs runner-up) -- inspect these first:\n")
print(as.data.frame(head(margin_df, 8)))
cat("\n")

write.csv(margin_df, file.path(output_dir, "cluster_annotation_margins.csv"), row.names = FALSE)
cat("Saved: cluster_annotation_margins.csv\n\n")

# ==============================================================================
# PLOT HEATMAP
# ==============================================================================

cat("Building heatmap...\n")

png(file.path(figure_dir, "Figure_S1_Cluster_Marker_Heatmap.png"),
    width = 10, height = 12, units = "in", res = 600)

pheatmap(
  score_mat_z,
  cluster_rows = TRUE,
  cluster_cols = TRUE,
  color = colorRampPalette(c("#2c7bb6", "white", "#d7191c"))(100),
  main = "Cluster-level marker module scores (z-scored per signature)",
  fontsize_row = 8,
  fontsize_col = 10,
  border_color = "grey80",
  angle_col = 45
)

dev.off()

pdf(file.path(figure_dir, "Figure_S1_Cluster_Marker_Heatmap.pdf"), width = 10, height = 12)
pheatmap(
  score_mat_z,
  cluster_rows = TRUE,
  cluster_cols = TRUE,
  color = colorRampPalette(c("#2c7bb6", "white", "#d7191c"))(100),
  main = "Cluster-level marker module scores (z-scored per signature)",
  fontsize_row = 8,
  fontsize_col = 10,
  border_color = "grey80",
  angle_col = 45
)
dev.off()

cat("Saved: Figure_S1_Cluster_Marker_Heatmap.png / .pdf\n\n")

cat(strrep("=", 100), "\n")
cat("QC DIAGNOSTIC COMPLETE\n")
cat(strrep("=", 100), "\n\n")

cat("Review 'cluster_annotation_margins.csv' -- any cluster with a small margin\n")
cat("(winner score close to runner-up score) may represent an ambiguous or\n")
cat("intermediate cell state, or possible doublets, and is worth a closer look.\n\n")
