cat('Preparing expression matrix for UCell...

expr_mat <- as.matrix(GetAssayData(scRNA, assay = 'RNA', layer = 'data'))

symbol_vec <- gene_map[rownames(expr_mat)]

dup_symbols <- symbol_vec[duplicated(symbol_vec) | duplicated(symbol_vec, fromLast = TRUE)]
if (length(dup_symbols) > 0) {
  cat('Found', length(unique(dup_symbols)), 'duplicate gene symbols. Making unique...
')
  dup_names <- names(dup_symbols)
  symbol_vec[dup_names] <- make.names(symbol_vec[dup_names], unique = TRUE)
}

rownames(expr_mat) <- symbol_vec

if (any(duplicated(rownames(expr_mat)))) {
  stop('Duplicate rownames still exist after make.names(). Check gene mapping.')
} else {
  cat('All rownames are unique. Proceeding with UCell scoring.
')
}

cat('Running UCell scoring on expression matrix (chunked to avoid memory limit)...
')

# Process in chunks of 5000 cells to stay under memory limit
chunk_size <- 5000
n_cells <- ncol(expr_mat)
cell_names <- colnames(expr_mat)
all_scores <- NULL

for (i in seq(1, n_cells, by = chunk_size)) {
  end_idx <- min(i + chunk_size - 1, n_cells)
  cat(sprintf('  Processing cells %d to %d of %d...
', i, end_idx, n_cells))
  
  chunk_mat <- expr_mat[, i:end_idx, drop = FALSE]
  chunk_scores <- ScoreSignatures_UCell(
    chunk_mat,
    features = gene_sets_symbols,
    maxRank = 1500,
    name = '_UCell'
  )
  
  if (is.null(all_scores)) {
    all_scores <- chunk_scores
  } else {
    all_scores <- rbind(all_scores, chunk_scores)
  }
  
  rm(chunk_mat, chunk_scores)
  gc()
}

ucell_scores <- all_scores
rm(all_scores, expr_mat)

run_ucell_corr <- function(df, group_label) {
  early <- cor.test(df$CUEDC2_expr, df$ER_Early_UCell, method = 'spearman', exact = FALSE)
  late  <- cor.test(df$CUEDC2_expr, df$ER_Late_UCell,  method = 'spearman', exact = FALSE)
  tibble(
    Method = paste0('UCell_', group_label),
    N_cells = nrow(df),   # FIXED: was N = nrow(df)
    Early_rho = unname(early$estimate), Early_P = early$p.value,
    Late_rho  = unname(late$estimate),  Late_P  = late$p.value
  )
}

ucell_per_sample <- ucell_meta %>%
  group_split(sample) %>%
  map_dfr(~ run_ucell_corr(.x, unique(.x$sample)))

ucell_results <- bind_rows(ucell_pooled_corr, ucell_per_sample) %>%
  mutate(
    Early_FDR = p.adjust(Early_P, method = 'BH'),
    Late_FDR  = p.adjust(Late_P,  method = 'BH')
  )

run_ucell_wilcox <- function(df, group_label) {
  med <- median(df$CUEDC2_expr)
  df$CUEDC2_group <- ifelse(df$CUEDC2_expr > med, 'High', 'Low')
  df <- df %>% filter(CUEDC2_group %in% c('High', 'Low'))
  
  early_w <- wilcox.test(ER_Early_UCell ~ CUEDC2_group, data = df)
  late_w  <- wilcox.test(ER_Late_UCell  ~ CUEDC2_group, data = df)
  
  tibble(
    Method = paste0('UCell_', group_label),
    N = nrow(df),
    N_High = sum(df$CUEDC2_group == 'High'),
    N_Low  = sum(df$CUEDC2_group == 'Low'),
    Median_Early_High = median(df$ER_Early_UCell[df$CUEDC2_group == 'High']),
    Median_Early_Low  = median(df$ER_Early_UCell[df$CUEDC2_group == 'Low']),
    Early_P = early_w$p.value,
    Median_Late_High = median(df$ER_Late_UCell[df$CUEDC2_group == 'High']),
    Median_Late_Low  = median(df$ER_Late_UCell[df$CUEDC2_group == 'Low']),
    Late_P = late_w$p.value
  )
}

ucell_wilcox_pooled <- run_ucell_wilcox(ucell_meta, 'Pooled')

  group_split(sample) %>%
  map_dfr(~ run_ucell_wilcox(.x, unique(.x$sample)))
